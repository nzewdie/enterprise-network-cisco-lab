! ==============================================================================
! ACCESS SWITCH 2 (AC2) RUNNING CONFIGURATION
! Enterprise Network Multi-Switch Enterprise Lab
! ==============================================================================
version 15.0
hostname AC2

! --- Access Layer Switching Services ---
spanning-tree mode pvst
spanning-tree extend system-id

! --- User Access Interface Ports (Edge Boundary) ---
interface FastEthernet0/1
 description Access Link to PC-3 (Data VLAN 10)
 switchport mode access
 switchport access vlan 10

interface FastEthernet0/2
 description Access Link to PC-4 (HR/Voice VLAN 20)
 switchport mode access
 switchport access vlan 20

! --- Core Infrastructure Uplinks (Redundant Trunks) ---
interface GigabitEthernet0/1
 description Redundant 802.1Q Uplink Trunk to CS1
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,99

interface GigabitEthernet0/2
 description Redundant 802.1Q Uplink Trunk to CS2
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,99

! --- Default VLAN Settings ---
interface Vlan1
 no ip address
 shutdown

! --- Console & Virtual Terminal Line Management ---
line con 0
 logging synchronous

line vty 0 4
 login

line vty 5 15
 login

end
